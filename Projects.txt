import { clsx, type ClassValue } from "clsx";
import { twMerge } from "tailwind-merge";

export function cn(...inputs: ClassValue[]) {
  return twMerge(clsx(inputs));
}

function RevealImagesShowcase() {
  const [selectedVariant, setSelectedVariant] = useState(0);

  const variants = [
    { component: <Variation1 />, label: "1" },
    { component: <Variation2 />, label: "2" },
    { component: <Variation3 />, label: "3" },
    { component: <Variation4 />, label: "4" },
    { component: <Variation5 />, label: "5" },
  ];

  return (
    <div className="flex flex-col md:flex-row h-screen w-full bg-background">
      {/* Mobile: Top selector */}
      <div className="md:hidden flex gap-2 p-4 overflow-x-auto border-b border-border bg-background">
        {variants.map((_, index) => (
          <button
            key={index}
            onClick={() => setSelectedVariant(index)}
            className={cn(
              "flex-shrink-0 relative w-20 h-20 rounded-lg border-2 overflow-hidden transition-all",
              selectedVariant === index
                ? "border-primary ring-2 ring-primary"
                : "border-border hover:border-primary/50"
            )}
          >
            <div className="absolute inset-0 pointer-events-none">
              <div
                style={{
                  transform: "scale(0.08)",
                  transformOrigin: "top left",
                  width: "1250%",
                  height: "1250%",
                }}
              >
                {variants[index].component}
              </div>
            </div>
            <div className="absolute bottom-1 right-1 bg-background/80 text-foreground text-xs font-bold px-1.5 py-0.5 rounded">
              {variants[index].label}
            </div>
          </button>
        ))}
      </div>

      {/* Main stage */}
      <div className="flex-1 flex items-center justify-center overflow-auto p-4 md:p-8">
        {variants[selectedVariant].component}
      </div>

      {/* Desktop: Right sidebar */}
      <div className="hidden md:flex flex-col gap-3 p-4 border-l border-border bg-background w-32">
        {variants.map((_, index) => (
          <button
            key={index}
            onClick={() => setSelectedVariant(index)}
            className={cn(
              "relative w-full h-24 rounded-lg border-2 overflow-hidden transition-all hover:scale-105",
              selectedVariant === index
                ? "border-primary ring-2 ring-primary"
                : "border-border hover:border-primary/50"
            )}
          >
            <div className="absolute inset-0 pointer-events-none">
              <div
                style={{
                  transform: "scale(0.12)",
                  transformOrigin: "top left",
                  width: "833%",
                  height: "833%",
                }}
              >
                {variants[index].component}
              </div>
            </div>
            <div className="absolute bottom-1 right-1 bg-background/80 text-foreground text-xs font-bold px-2 py-1 rounded">
              {variants[index].label}
            </div>
          </button>
        ))}
      </div>
    </div>
  );
}

export default RevealImagesShowcase;
